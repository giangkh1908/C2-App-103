## Backend Implementation Details

### `llm-stats` endpoint logic

```python
@router.get("/llm-stats")
async def get_llm_stats(
    days: int = Query(7, ge=1, le=90),
    admin=Depends(get_current_admin),
):
    db = get_db()
    now = datetime.now(UTC)
    since = now - timedelta(days=days)

    pipeline = [
        {"$match": {"created_at": {"$gte": since}}},
        {"$facet": {
            "daily_costs": [
                {"$group": {
                    "_id": {"$dateToString": {"format": "%Y-%m-%d", "date": "$created_at"}},
                    "cost_usd": {"$sum": "$cost_usd"},
                    "requests": {"$sum": 1},
                    "tokens": {"$sum": {"$add": ["$tokens_in", "$tokens_out"]}}
                }},
                {"$sort": {"_id": 1}}
            ],
            "cost_by_model": [
                {"$group": {
                    "_id": "$model",
                    "cost_usd": {"$sum": "$cost_usd"}
                }},
                {"$sort": {"cost_usd": -1}}
            ],
            "overall": [
                {"$group": {
                    "_id": None,
                    "total_cost_usd": {"$sum": "$cost_usd"},
                    "total_requests": {"$sum": 1},
                    "total_errors": {"$sum": {"$cond": [{"$eq": ["$status", "failure"]}, 1, 0]}},
                    "latencies": {"$push": "$latency_ms"}
                }}
            ]
        }}
    ]
    # ... compute p50/p95 from latencies array in code
```

### Budget alert job

```python
async def check_llm_budget():
    db = get_db()
    today_start = datetime.now(UTC).replace(hour=0, minute=0, second=0, microsecond=0)
    pipeline = [
        {"$match": {"created_at": {"$gte": today_start}}},
        {"$group": {"_id": None, "total": {"$sum": "$cost_usd"}}}
    ]
    result = await db.llm_audit_logs.aggregate(pipeline).to_list(1)
    today_cost = result[0]["total"] if result else 0.0

    budget = settings.llm_daily_budget_usd
    if today_cost > budget:
        # Find admin emails
        admins = await db.users.find({"role": "admin"}, {"email": 1}).to_list(...)
        for admin in admins:
            await send_email(admin["email"], "LLM Budget Alert", html_body)
        logger.warning("llm_budget_exceeded", daily_cost=today_cost, budget=budget)
```
